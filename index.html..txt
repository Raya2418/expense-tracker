      input.value = existingCols[i] || defaults[i] || `Field ${i + 1}`;
      input.dataset.colIndex = i;
      container.appendChild(input);
    }
  }

  function saveGridConfiguration() {
    const colCount = parseInt(document.getElementById('numColsInput').value) || 0;
    const rowCount = parseInt(document.getElementById('numRowsInput').value) || 0;
    
    if (colCount <= 0 || rowCount <= 0) {
      alert('Please enter a valid number greater than 0 for both rows and columns.');
      return;
    }

    const inputs = document.querySelectorAll('#columnInputsContainer input');
    const cols = Array.from(inputs).map(inp => inp.value.trim()).filter(v => v !== '');

    const vehicle = vehicles.find(v => v.id === activeVehicleId);
    vehicle.columns = cols;
    vehicle.rowCount = rowCount;
    vehicle.isConfigured = true;
    persist();

    closeModal('gridConfigModal');
    openLedgerScreen(vehicle);
  }

  function openLedgerScreen(vehicle) {
    document.getElementById('dashboard-screen').style.display = 'none';
    document.getElementById('ledger-screen').style.display = 'flex';
    document.getElementById('backBtn').style.display = 'block';
    document.getElementById('gridConfigBtn').style.display = 'block';
    document.getElementById('dashboardMenuOptions').style.display = 'none';
    document.getElementById('ledgerMenuOptions').style.display = 'block';

    document.getElementById('topTitle').innerText = vehicle.regNo;
    document.getElementById('topSubtitle').innerText = `${vehicle.columns.join(' • ')}`;
    
    const topDp = document.getElementById('topDp');
    topDp.innerHTML = vehicle.dp ? `<img src="${vehicle.dp}">` : '🚛';

    renderMessages(vehicle);
  }

  function showDashboard() {
    activeVehicleId = null;
    document.getElementById('dashboard-screen').style.display = 'flex';
    document.getElementById('ledger-screen').style.display = 'none';
    document.getElementById('backBtn').style.display = 'none';
    document.getElementById('gridConfigBtn').style.display = 'none';
    document.getElementById('dashboardMenuOptions').style.display = 'block';
    document.getElementById('ledgerMenuOptions').style.display = 'none';
    
    document.getElementById('topDp').innerHTML = '🚛';
    applyLanguage();
    renderVehicleList();
  }

  function renderMessages(vehicle) {
    const container = document.getElementById('messagesContainer');
    container.innerHTML = '';
    
    const numericTotals = {};
    vehicle.columns.forEach(col => numericTotals[col] = 0);

    vehicle.entries.forEach(entry => {
      const bubble = document.createElement('div');
      bubble.className = 'bubble';

      let rowsHtml = '';
      vehicle.columns.forEach(col => {
        const val = entry.data[col] || '-';
        rowsHtml += `
          <div class="bubble-row">
            <span class="bubble-key">${col}:</span>
            <span class="bubble-val">${val}</span>
          </div>
        `;
        const num = parseFloat(val);
        if (!isNaN(num)) numericTotals[col] += num;
      });

      bubble.innerHTML = `
        ${rowsHtml}
        <div class="bubble-bottom">
          <span>${entry.time || ''} ✓✓</span>
          <div class="bubble-actions">
            <button class="bubble-btn" onclick="openEditEntryModal('${entry.id}')">✏️</button>
            <button class="bubble-btn" onclick="deleteEntry('${entry.id}')">🗑️</button>
          </div>
        </div>
      `;
      container.appendChild(bubble);
    });

    const totalsArr = [];
    vehicle.columns.forEach(col => {
      if (numericTotals[col] > 0) totalsArr.push(`${col}: ${numericTotals[col].toFixed(1)}`);
    });
    document.getElementById('totalsRow').innerText = totalsArr.length > 0 ? `Totals: ${totalsArr.join(' | ')}` : 'Logged: ' + vehicle.entries.length + '/' + vehicle.rowCount;

    container.scrollTop = container.scrollHeight;
  }

  function openAddEntryModal() {
    editingEntryId = null;
    const vehicle = vehicles.find(v => v.id === activeVehicleId);
    if (vehicle.entries.length >= vehicle.rowCount) {
      if (!confirm(`Configured limit (${vehicle.rowCount} rows) reached. Add anyway?`)) return;
    }

    document.getElementById('entryModalTitle').innerText = I18N[currentLang].addEntry;
    buildEntryForm({});
    openModal('entryModal');
  }

  function openEditEntryModal(entryId) {
    editingEntryId = entryId;
    const vehicle = vehicles.find(v => v.id === activeVehicleId);
    const entry = vehicle.entries.find(e => e.id === entryId);
    
    document.getElementById('entryModalTitle').innerText = I18N[currentLang].editEntry;
    buildEntryForm(entry.data);
    openModal('entryModal');
  }

  function buildEntryForm(data) {
    const vehicle = vehicles.find(v => v.id === activeVehicleId);
    const container = document.getElementById('entryInputsContainer');
    container.innerHTML = '';

    vehicle.columns.forEach(col => {
      const input = document.createElement('input');
      input.className = 'input-field';
      input.id = `input_entry_${col}`;
      input.placeholder = col;
      if (col.toLowerCase().includes('date')) {
        input.type = 'date';
        input.value = data[col] || new Date().toISOString().split('T')[0];
      } else {
        input.type = (col.toLowerCase().includes('cost') || col.toLowerCase().includes('amount') || col.toLowerCase().includes('litre') || col.toLowerCase().includes('bata')) ? 'number' : 'text';
        input.value = data[col] || '';
      }
      container.appendChild(input);
    });
  }

  function saveEntryForm() {
    const vehicle = vehicles.find(v => v.id === activeVehicleId);
    const dataObj = {};
    
    vehicle.columns.forEach(col => {
      const val = document.getElementById(`input_entry_${col}`).value.trim();
      dataObj[col] = val || '-';
    });

    const now = new Date();
    const timeStr = now.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });

    if (editingEntryId) {
      const idx = vehicle.entries.findIndex(e => e.id === editingEntryId);
      if (idx !== -1) vehicle.entries[idx].data = dataObj;
    } else {
      vehicle.entries.push({ id: Date.now().toString(), data: dataObj, time: timeStr });
    }

    persist();
    closeModal('entryModal');
    renderMessages(vehicle);
  }

  function deleteEntry(entryId) {
    if (!confirm('Delete this entry?')) return;
    const vehicle = vehicles.find(v => v.id === activeVehicleId);
    vehicle.entries = vehicle.entries.filter(e => e.id !== entryId);
    persist();
    renderMessages(vehicle);
  }

  function openAddVehicleModal() {
    document.getElementById('newRegInput').value = '';
    document.getElementById('newDpInput').value = '';
    openModal('addVehicleModal');
  }

  function saveNewVehicle() {
    const regNo = document.getElementById('newRegInput').value.trim().toUpperCase();
    if (!regNo) return;
    
    const file = document.getElementById('newDpInput').files[0];
    if (file) {
      const reader = new FileReader();
      reader.onload = (e) => {
        vehicles.push({ id: Date.now().toString(), regNo, dp: e.target.result, rowCount: 10, columns: [], isConfigured: false, entries: [] });
        persist();
        closeModal('addVehicleModal');
        renderVehicleList();
      };
      reader.readAsDataURL(file);
    } else {
      vehicles.push({ id: Date.now().toString(), regNo, dp: '', rowCount: 10, columns: [], isConfigured: false, entries: [] });
      persist();
      closeModal('addVehicleModal');
      renderVehicleList();
    }
  }

  function handleDpUpload(event) {
    const file = event.target.files[0];
    if (!file || !activeVehicleId) return;

    const reader = new FileReader();
    reader.onload = (e) => {
      const vehicle = vehicles.find(v => v.id === activeVehicleId);
      vehicle.dp = e.target.result;
      persist();
      document.getElementById('dpFullImg').src = vehicle.dp;
      document.getElementById('topDp').innerHTML = `<img src="${vehicle.dp}">`;
      renderVehicleList();
    };
    reader.readAsDataURL(file);
  }

  function exportCurrentVehicleExcel() {
    const vehicle = vehicles.find(v => v.id === activeVehicleId);
    if (!vehicle || vehicle.entries.length === 0) {
      alert('No entries to export.');
      return;
    }

    const data = vehicle.entries.map(e => e.data);
    const ws = XLSX.utils.json_to_sheet(data);
    const wb = XLSX.utils.book_new();
    XLSX.utils.book_append_sheet(wb, ws, vehicle.regNo.replace(/[^a-zA-Z0-9]/g, '_'));
    XLSX.writeFile(wb, `${vehicle.regNo}_Expense_Ledger.xlsx`);
  }

  function exportAllVehiclesExcel() {
    if (vehicles.length === 0) return;
    const wb = XLSX.utils.book_new();

    vehicles.forEach(v => {
      const data = v.entries.map(e => e.data);
      const ws = XLSX.utils.json_to_sheet(data.length > 0 ? data : [{ Info: 'No entries' }]);
      XLSX.utils.book_append_sheet(wb, ws, v.regNo.substring(0, 30));
    });

    XLSX.writeFile(wb, `All_Vehicles_Ledger_${new Date().toISOString().split('T')[0]}.xlsx`);
  }
