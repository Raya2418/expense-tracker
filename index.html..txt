  function compressImage(file, callback) {
    const reader = new FileReader();
    reader.onload = function(e) {
      const img = new Image();
      img.onload = function() {
        const canvas = document.createElement('canvas');
        const MAX_DIM = 300;
        let width = img.width;
        let height = img.height;

        if (width > height) {
          if (width > MAX_DIM) {
            height = Math.round((height * MAX_DIM) / width);
            width = MAX_DIM;
          }
        } else {
          if (height > MAX_DIM) {
            width = Math.round((width * MAX_DIM) / height);
            height = MAX_DIM;
          }
        }

        canvas.width = width;
        canvas.height = height;
        const ctx = canvas.getContext('2d');
        ctx.drawImage(img, 0, 0, width, height);
        callback(canvas.toDataURL('image/jpeg', 0.65));
      };
      img.src = e.target.result;
    };
    reader.readAsDataURL(file);
  }

  function cleanNumber(val) {
    if (val === null || val === undefined) return 0;
    const cleaned = String(val).replace(/[^0-9.-]+/g, '');
    const num = parseFloat(cleaned);
    return isNaN(num) ? 0 : num;
  }

  function renderColumnNameInputs() {
    const colCount = parseInt(document.getElementById('numColsInput').value) || 0;
    const container = document.getElementById('columnInputsContainer');
    container.innerHTML = '<label style="font-size:12px; color:var(--wa-text-secondary); margin:8px 0 4px 0; display:block;">Column Names:</label>';

    if (colCount <= 0) return;

    const vehicle = vehicles.find(v => v.id === activeVehicleId);
    const existingCols = vehicle ? vehicle.columns : [];
    const defaults = I18N[currentLang].defaults;

    for (let i = 0; i < colCount; i++) {
      const input = document.createElement('input');
      input.className = 'input-field';
      input.placeholder = `Column ${i + 1} Name`;
      input.value = (existingCols[i] ? existingCols[i].name : '') || defaults[i] || `Field ${i + 1}`;
      input.dataset.colIndex = i;
      container.appendChild(input);
    }
  }

  function saveGridConfiguration() {
    const colCount = parseInt(document.getElementById('numColsInput').value) || 0;
    const rowCount = parseInt(document.getElementById('numRowsInput').value) || 0;

    if (colCount <= 0 || rowCount <= 0) {
      alert('Please enter a valid number greater than 0 for both columns and rows.');
      return;
    }

    const inputs = document.querySelectorAll('#columnInputsContainer input');
    const vehicle = vehicles.find(v => v.id === activeVehicleId);
    const updatedCols = [];

    Array.from(inputs).forEach((inp, idx) => {
      const name = inp.value.trim() || `Field ${idx + 1}`;
      const existingCol = vehicle.columns[idx];
      updatedCols.push({
        id: existingCol ? existingCol.id : 'col_' + Date.now() + '_' + idx,
        name: name
      });
    });

    vehicle.columns = updatedCols;
    vehicle.rowCount = rowCount;
    vehicle.isConfigured = true;
    persist();

    closeModal('gridConfigModal');
    openLedgerScreen(vehicle);
  }

  function renderMessages(vehicle) {
    const container = document.getElementById('messagesContainer');
    container.innerHTML = '';

    const numericTotals = {};
    vehicle.columns.forEach(col => numericTotals[col.id] = 0);

    if (vehicle.entries.length === 0) {
      container.innerHTML = `<div style="text-align:center; padding:30px 20px; color:#667781; font-size:13px;">No expense records logged yet.<br>Tap "Add Entry" below.</div>`;
    }

    vehicle.entries.forEach(entry => {
      const bubble = document.createElement('div');
      bubble.className = 'bubble';

      let rowsHtml = '';
      vehicle.columns.forEach(col => {
        const val = entry.data[col.id] || '-';
        rowsHtml += `
          <div class="bubble-row">
            <span class="bubble-key">${col.name}:</span>
            <span class="bubble-val">${val}</span>
          </div>
        `;
        numericTotals[col.id] += cleanNumber(val);
      });

      bubble.innerHTML = `
        ${rowsHtml}
        <div class="bubble-bottom">
          <span>${entry.time || ''} <span style="color:#53bdeb; font-weight:bold;">✓✓</span></span>
          <div class="bubble-actions">
            <button class="bubble-btn" title="Edit this entry" onclick="openEditEntryModal('${entry.id}')">✏️</button>
            <button class="bubble-btn" title="Delete this entry" onclick="deleteEntry('${entry.id}')">🗑️</button>
          </div>
        </div>
      `;
      container.appendChild(bubble);
    });

    const totalsArr = [];
    vehicle.columns.forEach(col => {
      if (numericTotals[col.id] > 0) {
        totalsArr.push(`${col.name}: ${numericTotals[col.id].toLocaleString('en-IN')}`);
      }
    });

    document.getElementById('totalsRow').innerText = totalsArr.length > 0 
      ? `Totals: ${totalsArr.join(' | ')}` 
      : `Logged: ${vehicle.entries.length}/${vehicle.rowCount} entries`;

    container.scrollTop = container.scrollHeight;
  }

  function openEditEntryModal(entryId) {
    editingEntryId = String(entryId);
    const vehicle = vehicles.find(v => v.id === activeVehicleId);

  function buildEntryForm(data) {
    const vehicle = vehicles.find(v => v.id === activeVehicleId);
    const container = document.getElementById('entryInputsContainer');
    container.innerHTML = '';

    vehicle.columns.forEach(col => {
      const label = document.createElement('label');
      label.style.fontSize = '12px';
      label.style.color = 'var(--wa-text-secondary)';
      label.innerText = col.name;
      container.appendChild(label);

      const input = document.createElement('input');
      input.className = 'input-field';
      input.id = `input_entry_${col.id}`;
      input.placeholder = col.name;

      const n = col.name.toLowerCase();
      if (n.includes('date') || col.name.includes('ದಿನಾಂಕ') || col.name.includes('तारीख')) {
        input.type = 'date';
        input.value = data[col.id] || new Date().toISOString().split('T')[0];
      } else if (n.includes('cost') || n.includes('amount') || n.includes('litre') || n.includes('bata') || n.includes('price') || col.name.includes('₹') || col.name.includes('ಲೀಟರ್') || col.name.includes('लीटर')) {
        input.type = 'number';
        input.step = 'any';
        input.value = (data[col.id] !== undefined && data[col.id] !== '-') ? data[col.id] : '';
      } else {
        input.type = 'text';
        input.value = (data[col.id] !== undefined && data[col.id] !== '-') ? data[col.id] : '';
      }
      container.appendChild(input);
    });
  }

  function saveEntryForm() {
    const vehicle = vehicles.find(v => v.id === activeVehicleId);
    const dataObj = {};

    vehicle.columns.forEach(col => {
      const val = document.getElementById(`input_entry_${col.id}`).value.trim();
      dataObj[col.id] = val || '-';
    });

    const now = new Date();
    const timeStr = now.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });

    if (editingEntryId !== null) {
      const targetIndex = vehicle.entries.findIndex(e => String(e.id) === String(editingEntryId));
      if (targetIndex !== -1) {
        vehicle.entries[targetIndex].data = dataObj;
        vehicle.entries[targetIndex].time = timeStr + ' (edited)';
      }
    } else {
      vehicle.entries.push({ 
        id: 'entry_' + Date.now() + '_' + Math.random().toString(36).substr(2, 5), 
        data: dataObj, 
        time: timeStr 
      });
    }

    editingEntryId = null;
    persist();
    closeModal('entryModal');
    renderMessages(vehicle);
  }

  function deleteEntry(entryId) {
    if (!confirm('Delete this entry?')) return;
    const vehicle = vehicles.find(v => v.id === activeVehicleId);
    vehicle.entries = vehicle.entries.filter(e => String(e.id) !== String(entryId));
    persist();
    renderMessages(vehicle);
  }

  function sanitizeSheetName(name) {
    return name.replace(/[\\/?*\[\]:]/g, '_').substring(0, 30);
  }

  function exportCurrentVehicleExcel() {
    const vehicle = vehicles.find(v => v.id === activeVehicleId);
    if (!vehicle || vehicle.entries.length === 0) {
      alert('No entries to export.');
      return;
    }

    const exportRows = vehicle.entries.map(e => {
      const row = {};
      vehicle.columns.forEach(col => {
        row[col.name] = e.data[col.id] || '-';
      });
      return row;
    });

    const ws = XLSX.utils.json_to_sheet(exportRows);
    const wb = XLSX.utils.book_new();
    XLSX.utils.book_append_sheet(wb, ws, sanitizeSheetName(vehicle.regNo));
    XLSX.writeFile(wb, `${sanitizeSheetName(vehicle.regNo)}_Expenses.xlsx`);
  }

  function exportAllVehiclesExcel() {
    if (vehicles.length === 0) {
      alert('No vehicles to export.');
      return;
    }
    const wb = XLSX.utils.book_new();

    vehicles.forEach(v => {
      const exportRows = v.entries.map(e => {
        const row = {};
        v.columns.forEach(col => {
          row[col.name] = e.data[col.id] || '-';
        });
        return row;
      });

      const ws = XLSX.utils.json_to_sheet(exportRows.length > 0 ? exportRows : [{ Info: 'No entries logged' }]);
      XLSX.utils.book_append_sheet(wb, ws, sanitizeSheetName(v.regNo));
    });

    XLSX.writeFile(wb, `All_Vehicles_Expenses_${new Date().toISOString().split('T')[0]}.xlsx`);
  }
